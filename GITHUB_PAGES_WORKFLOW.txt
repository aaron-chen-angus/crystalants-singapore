Create this file in GitHub using Add file -> Create new file:

.github/workflows/deploy-pages.yml

Paste the following contents:

name: Build and deploy Crystal Ants

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Check out source
        uses: actions/checkout@v4

      - name: Install Godot 4.7.2
        shell: bash
        run: |
          curl -L --fail --retry 3 -o godot.zip \
            https://github.com/godotengine/godot/releases/download/4.7.2-stable/Godot_v4.7.2-stable_linux.x86_64.zip
          unzip -q godot.zip
          chmod +x Godot_v4.7.2-stable_linux.x86_64

      - name: Export Web build
        shell: bash
        run: |
          mkdir -p site
          ./Godot_v4.7.2-stable_linux.x86_64 \
            --headless --audio-driver Dummy --path . \
            --export-release Web site/index.html

      - name: Upload Pages artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: site

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
